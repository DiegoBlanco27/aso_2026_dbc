# Practica 2: Condicional ```if```
Vamos a realizar las siguentes tareas usando ```if``` en los scripts:

## Ejercicio 1: Contar lineas de un archivo
```
#!/bin/bash

read -p "Introduce el nombre del archivo: " archivo

if [ -f "$archivo" ]; then
    lineas=$(wc -l < "$archivo")
    
    echo "El archivo $archivo tiene $lineas líneas."
else
    echo "El archivo archivo no existe"
fi
```

## Ejercicio 2: Verificación de archivo
```
#!/bin/bash

read -p "Introduce la ruta del archivo: " archivo

if [ -e "$archivo" ] then
    echo "El archivo existe."

    if [ -r "$archivo" ] then
        echo "Tienes permisos de lectura sobre el archivo."
    else
        echo "No tienes permisos de lectura para este archivo."
    fi
else
    echo "El archivo no esta en esta ruta."
fi
```

## Ejercicio 3: Comparación de dos números
```
#!/bin/bash

read -p "Introduce un número: " num1
read -p "Introduce otro número: " num2

if [ $num1 -gt $num2 ]; then
    echo "El número $num1 es mayor que el número $num2"
elif [ $num2 -gt $num1 ]; then
    echo "El número $num2 es mayor que el número $num1"
else
    echo "Es el mismo número"
fi
```

## Ejercicio 4: Validación  de contraseñas
```
#!/bin/bash

read -p "Introduce la contraseña: " contrasena

password=paso

if
    [ $contrasena = password ]; then
    echo "Exito"
else

fi
```

## Ejercicio 5: Comprobación de directorio
```
#!/bin/bash

read -p "Introduce un directorio: " directorio

if
    [ -d $directorio ]; then
    echo "El directorio existe"
else
    echo "El directorio no existe, creando directorio $directorio"
    mkdir $directorio
fi

if
    [ -w $directorio ]; then
    echo "El directorio tiene permisos de escritura"
else
    echo "El directorio no tiene permisos de escritura"
fi
```

## Ejercicio 6: Verificar si el usuario es root
```
#!/bin/bash

if
        [ $UID -eq 0 ]; then
        echo "Hola, root"
else
        echo "No eres root"
fi
```

```
#!/bin/bash

```
## Ejercicio 8: Comprobación del espacio en disco
```
#!/bin/bash

PORCENTAJE=10
RUTA="/"

USADO=$(df $RUTA | tail -n 1 | awk '{print $5}' | tr -d '%')

ESPACIO=$((100 - USADO))

if [ $ESPACIO -lt $PORCENTAJE ]; then
    echo "ADVERTENCIA: El espacio libre en $RUTA es del $ESPACIO% solamente"
else
    echo "El espacio libre en $RUTA es del $ESPACIO%"
fi
```

## Ejercicio 9: Menú de opciones
```
#!/bin/bash

echo "Opción 1: Fecha"
echo "Opción 2: Usuario actual"
echo "Opcion 3: Salir"

read -p "Selecciona una opción: " opcion

case $opcion in
        1)
                date;;
        2)
                whoami;;
        3)      
                echo "Has salido del script";;
        *)
                echo "Opción no disponible";;
esac
```

## Ejercicio 10: Evaluación de edad
```
#!/bin/bash

read -p "Introduce tu edad: " edad

if [ $edad -lt 18 ]; then
    echo "Eres menor de edad"
elif [ $edad -le 65 ]; then
    echo "Eres adulto"
else
    echo "Eres mayor de edad"
fi
```
