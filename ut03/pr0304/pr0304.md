# Practica 4: El comando ```case``` 
## Ejercicio 1: Menú de operaciones matemáticas
   ```
    #!/bin/bash

    read -p "Introduce el primer número: " primero
    read -p "Introduce el segundo número: " segundo

    echo "¿Que quieres hacer? "
    echo  " 1) Sumar  2) Restar 3) Multiplicar 4) Dividir " 

    read operacion  

    case $operacion in
        1)
                resultado=$(( $primero + $segundo ))
                echo "$primero + $segundo = $resultado";;
        2)
                resultado=$(( $primero - $segundo ))
                echo "$primero - $segundo = $resultado";;
        3)
                resultado=$(( $primero * $segundo ))
                echo "$primero * $segundo = $resultado";;
        4)
                resultado=$(( $primero / $segundo ))
                echo "$primero / $segundo = $resultado";;
        5)
                echo "Selecciona una opcion valida";;
    esac
   ```

## Ejercicio 2: Verificar días de la semana
   ```
    #!/bin/bash

   read -p "Introduce un dia de la semana: " dia

    case $dia in
        lunes | martes | miercoles | jueves | viernes)
                echo "El $dia es un dia laboral";;
        sabado | domingo)
                echo "El $dia es fin de semana";;
        *)
                echo "$dia no es un dia de la semana";;
    esac 
   ```

## Ejercicio 4: Determinar la estación del año
   ```
    #!/bin/bash

    read -p "Introduce un mes: " mes
    read -p "Introduce un dia: " dia

    if [ $dia -gt 31 ]; then 
        echo "Ese dia no existe"
        exit 1
    fi
    case $mes in
        diciembre)
                if [ $dia -lt 21 ]; then  
                        echo "Es otoño"
                else 
                        echo "Es invierno"
                fi;; 
        enero | febrero)
                echo "Es invierno";;
        marzo)
                if [ $dia -lt 21 ]; then 
                        echo "Es invierno"
                else
                        echo "Es primavera"
                fi;;
        abril | mayo)
                echo "Es primavera";;
        junio)
                if [ $dia -lt 21 ]; then  
                        echo "Es primavera"
                else 
                        echo "Es verano"
                fi;; 
        julio | agosto)
                echo "Es verano";;
        septiembre)
                if [ $dia -lt 21 ]; then  
                        echo "Es verano"
                else 
                        echo "Es otoño"
                fi;; 
        octubre | noviembre)
                echo "Es otoño";;
        *)
                echo "Ese mes no existe";;
    esac
   ```

## Ejercicio 7: Estado del servicio
   ```
    #!/bin/bash

    read -p "Introduce un servicio: " servicio

    echo "Selecciona una opción:"
    echo "1. Iniciar servicio"
    echo "2. Detener servicio"
    echo "3. Reiniciar servicio"
    read opcion

    case $opcion in
        1)
                sudo systemctl start "$servicio"
                echo "Servicio $servicio iniciado."
                ;;
        2)
                sudo systemctl stop "$servicio"
                echo "Servicio $servicio detenido."
                ;;
        3)
                sudo systemctl restart "$servicio"
                echo "Servicio $servicio reiniciado."
                ;;
        *)
                echo "Opción no válida."
                ;;
        esac
   ```

## Ejercicio 8: Conversión de unidades de longitud
   ```
    #!/bin/bash

    read -p "Introduce la cantidad de metros: " metros
    read -p "¿A que unidad lo quieres pasar? (km, cm o mm) " unidad
    
    case $unidad in
        km)
                resultado=$(( $metros / 1000 ))
                echo "$resultado kilómetros.";;
        cm)
                resultado=$(( $metros * 100 ))
                echo "$resultado centímetros.";;
        mm)
                resultado=$(( $metros * 1000 ))
                echo "$resultado milímetros.";;
        *)
                echo "Opción no válida.";;
    esac
   ```
